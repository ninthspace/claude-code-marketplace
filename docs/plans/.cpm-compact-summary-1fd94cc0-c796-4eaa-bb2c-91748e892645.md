```bash
inspect_selector_direction() {
  local repo="$1"; shift
  [ $# -gt 0 ] || return 1
  case "$1" in
    --*)   printf 'git\n';    return 0 ;;
    *..*)  printf 'git\n';    return 0 ;;
  esac
  if [ $# -eq 1 ] && [ -d "$repo" ] \
     && git -C "$repo" show-ref --verify --quiet "refs/heads/$1"; then
    printf 'git\n'; return 0
  fi
  printf 'intent\n'
}

inspect_resolve() {
  local repo="$1"; shift
  if [ $# -eq 0 ]; then
    echo "inspect-resolve: no selector given" >&2; return 1
  fi
  local direction
  direction=$(inspect_selector_direction "$repo" "$@")
  case "$direction" in
    git)    changeset_resolve_git "$repo" "$@" ;;
    intent) changeset_resolve_intent "$repo" "$*" ;;
  esac
}
```
Header records: rule 3 resolves toward the branch because `show-ref` is checkable while "some adapter might understand this" is not; CPM ids carry a space so never reach it; it decides for issue keys like `AUTH-4`. The dispatcher adds **no** error path except "no selector at all" (which would otherwise return success having done nothing).

The character-scanning JSON string reader (handles `", ` inside criterion text, which defeats any punctuation split):
```awk
function qstrings(line, out,    i, n, c, inq, esc, cur, k) {
  n = length(line); inq = 0; esc = 0; cur = ""; k = 0
  for (i = 1; i <= n; i++) {
    c = substr(line, i, 1)
    if (!inq) { if (c == "\"") { inq = 1; cur = "" } ; continue }
    if (esc) {
      if (c == "n") cur = cur "\n"
      else if (c == "t") cur = cur "\t"
      else if (c == "r") cur = cur "\r"
      else if (c == "b") cur = cur sprintf("%c", 8)
      else if (c == "f") cur = cur sprintf("%c", 12)
      else cur = cur c          # covers \" and \\, which are their own literal
      esc = 0; continue
    }
    if (c == "\\") { esc = 1; continue }
    if (c == "\"") { out[++k] = cur; inq = 0; continue }
    cur = cur c
  }
  return k
}
```
Section detection keys on **key name AND indentation** — `files` appears twice (4-space under `changeset`, 2-space at top level) and a name-only reader would merge them.

### Created: `cpm/hooks/lib/inspect-run.sh` (Story 3)
The whole deterministic half of `/cpm:inspect` in one call.

```bash
INSPECT_LINK_ADAPTERS="${INSPECT_LINK_ADAPTERS-gitnative cpm}"   # `-` not `:-`, so empty is honoured

inspect_run() {
  local repo="$1" work="$2" budget="$3"; shift 3
  # guards: repo dir, work dir, $# -eq 0
  mkdir -p "$work" || return 1
  local selector="$*"
  local changeset="$work/changeset" linkset="$work/linkset" gaps="$work/gaps"
  local selection="$work/selection" payload="$work/payload"

  local direction
  direction=$(inspect_selector_direction "$repo" "$@") || return 1
  inspect_resolve "$repo" "$@" > "$changeset" || return 1

  linkset_reset
  local name registered=""
  for name in $INSPECT_LINK_ADAPTERS; do
    if declare -f "${name}_link_changeset" >/dev/null 2>&1; then
      linkset_register "$name" || return 1
      registered="${registered}${registered:+ }$name"
    fi
  done
  linkset_join "$repo" "$changeset" > "$linkset" || return 1
  gap_report "$repo" "$changeset" < "$linkset" > "$gaps" || return 1
  review_select "$changeset" "$budget" < "$linkset" > "$selection" || return 1
  review_payload "$changeset" < "$linkset" > "$payload" || return 1

  local record
  record=$(inspect_write "$repo" "$changeset" "$selector" < "$linkset") || return 1

  printf 'SELECTOR\t%s\n' "$selector"
  printf 'DIRECTION\t%s\n' "$direction"
  for name in $registered; do printf 'ADAPTER\t%s\n' "$name"; done
  printf 'COMMITS\t%s\n' "$(changeset_commits < "$changeset" | grep -c '[^[:space:]]')"
  printf 'FILES\t%s\n' "$(changeset_files < "$changeset" | grep -c '[^[:space:]]')"
  printf 'CHANGESET\t%s\n' "$changeset"
  printf 'LINKSET\t%s\n' "$linkset"
  printf 'GAPS\t%s\n' "$gaps"
  printf 'SELECTION\t%s\n' "$selection"
  printf 'PAYLOAD\t%s\n' "$payload"
  printf 'RECORD\t%s\n' "$record"
}
```
Sources six libs via individual `if ! declare -f X; then source Y; fi` blocks (matching every other lib here). **The record is written last** so a run that fails part-way leaves nothing behind claiming to describe it.
